// backend/index.js

const express = require('express');
const bodyParser = require('body-parser');
const cors = require('cors');

const app = express();
const PORT = process.env.PORT || 5000;

// Middleware
app.use(cors());
app.use(bodyParser.json());

// Simple endpoint to create a laundry request
app.post('/create-order', (req, res) => {
  const { userId, laundryItems, serviceType } = req.body;

  // Here, you can add logic to save the order in a database
  console.log(`Order created for User: ${userId}, Laundry Items: ${laundryItems}, Service Type: ${serviceType}`);

  res.status(201).send({ message: 'Order created successfully!' });
});

// Start the server
app.listen(PORT, () => {
  console.log(`Server is running on port ${PORT}`);
});
